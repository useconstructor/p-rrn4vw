# Fetch raw HTML to extract subpage links
curl -sL --max-time 15 -A "Mozilla/5.0" "https://cal.com" -o /tmp/pages/home.html 2>&1

# Extract internal links from raw HTML
cat << 'EXTRACTEOF' > /tmp/extract-links.js
const fs = require('fs');
const html = fs.readFileSync('/tmp/pages/home.html', 'utf8');
const base = 'https://cal.com';
const seen = new Set(['/']);
const links = [];
const re = /href="([^"]+)"/g;
let m;
while ((m = re.exec(html)) !== null) {
  const href = m[1];
  try {
    const u = new URL(href, base);
    if (u.origin === base && u.pathname !== '/' && !seen.has(u.pathname) && !u.pathname.match(/\.(png|jpg|pdf|svg|ico|js|css)$/) && !u.pathname.includes('#')) {
      seen.add(u.pathname);
      links.push(u.pathname);
      if (links.length >= 8) break;
    }
  } catch(e) {}
}
console.log(JSON.stringify(links));
EXTRACTEOF
SUBPAGES=$(node /tmp/extract-links.js 2>/dev/null || echo "[]")
echo "Subpages found: $SUBPAGES"
echo "$SUBPAGES" > /tmp/pages/subpages.json

# Fetch content for each subpage
for SLUG in $(echo "$SUBPAGES" | node -e "const d=require('fs').readFileSync('/dev/stdin','utf8'); try{JSON.parse(d).forEach(p=>console.log(p))}catch(e){}"); do

  echo "Fetched $SLUG: $(wc -c < "/tmp/pages/$FILENAME.md") bytes"
done

echo "=== Available pages ==="
